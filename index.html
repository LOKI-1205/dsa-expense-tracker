// DJB2 Hash Algorithm Implementation
hash(key) {
  let hash = 5381;
  for (let i = 0; i < key.length; i++) {
    hash = (hash * 33) ^ key.charCodeAt(i);
  }
  return (hash >>> 0) % this.capacity;
}

// Dynamic Capacity Expansion
push(element) {
  if (this.size === this.capacity) {
    this._resize(this.capacity * 2);
  }
  this.buffer[this.size++] = element;
}

// Heapify Sift-Down Logic
siftDown(i) {
  let maxIdx = i;
  let left = 2 * i + 1, right = 2 * i + 2;
  if (left < size && arr[left] > arr[maxIdx]) maxIdx = left;
  if (right < size && arr[right] > arr[maxIdx]) maxIdx = right;
  if (i !== maxIdx) {
    swap(i, maxIdx);
    siftDown(maxIdx);
  }
}
